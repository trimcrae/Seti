    % body

      .. controls (0.6,0.9) and (0.9,-0.2) .. (0.7,-1.0) -- cycle;
    % belly
    \fill[white] (0,-0.2) ellipse (0.45 and 0.7);

    \fill[black] (-0.9,-0.2) .. controls (-1.4,0.2) and (-1.1,-0.6) .. (-0.7,-0.9) -- cycle;

    \fill[white] ( 0.18,0.75) circle (0.12);
    \fill[black] (-0.18,0.75) circle (0.05);
    \fill[black] ( 0.18,0.75) circle (0.05);
    % beak & feet (set to black for monochrome)

    \fill[orange] ( 0.30,-1.05) -- ( 0.60,-1.25) -- ( 0.10,-1.25) -- cycle;
  \end{tikzpicture}%
}
